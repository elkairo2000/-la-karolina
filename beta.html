const STORAGE_KEY = "laKarolinaBeta2"; let products = []; function load() { try { products = JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; } catch (error) { products = []; } } function save() { localStorage.setItem(STORAGE_KEY, JSON.stringify(products)); } function show(id) { document.querySelectorAll("section").forEach(section => { section.hidden = section.id !== id; }); if (id === "inventario") { render(); } } function addProduct() { const name = document.getElementById("nombre").value.trim(); const shelf = document.getElementById("estante").value.trim(); const qty = Number(document.getElementById("cantidad").value); const cost = Number(document.getElementById("costo").value); if (!name || document.getElementById("cantidad").value === "" || document.getElementById("costo").value === "" || !Number.isInteger(qty) || qty < 0 || !Number.isFinite(cost) || cost < 0) { alert("Completa correctamente los datos"); return; } products.push({ id: Date.now() + Math.random(), name, shelf, qty, cost }); save(); document.getElementById("nombre").value = ""; document.getElementById("estante").value = ""; document.getElementById("cantidad").value = ""; document.getElementById("costo").value = ""; render(); show("inventario"); alert("Producto registrado correctamente"); } function render() { const list = document.getElementById("listaProductos"); const search = document.getElementById("buscar").value.toLowerCase(); list.innerHTML = ""; const filtered = products.filter(product => product.name.toLowerCase().includes(search) ); if (filtered.length === 0) { list.textContent = products.length === 0 ? "Todavía no hay productos registrados." : "No se encontraron productos."; return; } filtered.forEach(product => { const div = document.createElement("div"); div.className = "item"; div.textContent = product.name + " | Estante: " + (product.shelf || "Sin asignar") + " | Cantidad: " + product.qty + " | Costo: " + product.cost; if (product.qty <= 2) { div.classList.add("warning"); div.textContent += " ⚠️ EXISTENCIAS BAJAS"; } list.appendChild(div); }); } load(); render();
